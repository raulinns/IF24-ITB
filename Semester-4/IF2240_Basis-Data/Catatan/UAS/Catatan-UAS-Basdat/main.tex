\section{Capaian Pembelajaran Mata Kuliah (CPMK)}

\begin{enumerate}
    \item Menjelaskan peranan sistem basis data dalam pemenuhan kebutuhan akan informasi.
    \item Menjelaskan model data relasional dan mengekspresikan kebutuhan data menggunakan operasi relasional.
    \item Melakukan pemodelan data skala kecil-menengah dengan menggunakan model entity-relationship.
    \item Membuat rancangan skema basis data relasional.
    \item Mengimplementasikan sebuah basis data menggunakan DBMS relasional.
    \item Menyusun perintah SQL untuk memperoleh data dan informasi dari basis data serta memanipulasi data di dalam basis data.
\end{enumerate}

\section{Outline Keseluruhan Materi}

Bagian ini menyajikan peta belajar ringkas untuk seluruh materi yang diujikan.
Penjelasan mendalam tersedia pada section masing-masing materi.

\subsection{Introduction to Database}
\begin{itemize}
    \item Data, basis data, DBMS, peran DBMS, abstraksi data, schema-instance, data independence, komponen DBMS, dan bahasa basis data.
    \item Proses pengembangan basis data melalui data modeling, SDLC, dan prototyping/RAD.
\end{itemize}

\subsection{Model Data}
\begin{itemize}
    \item Tujuan pemodelan, level conceptual-logical-physical, dan prosedur identifikasi grup data.
    \item Kategori model data: E-R, relational, semi-structured, object-based, network, hierarchical, dan hubungan antar model.
\end{itemize}

\subsection{Relational Model}
\begin{itemize}
    \item Relation, attribute, tuple, domain, atomic value, schema, instance, key, foreign key, null, dan integrity constraints.
    \item Relational model menjadi dasar SQL, aljabar relasional, kalkulus relasional, dan normalisasi.
\end{itemize}

\subsection{Relational Algebra \& Calculus}
\begin{itemize}
    \item Operator dasar aljabar relasional: selection, projection, union, set difference, Cartesian product, dan rename.
    \item Operator turunan, aggregation, null, multiset semantics, tuple relational calculus, domain relational calculus, dan relational completeness.
\end{itemize}

\subsection{SQL (Structured Query Language)}
\begin{itemize}
    \item DDL, DML, query dasar, null logic, set operations, aggregation, grouping, joins, nested subquery, views, modification, authorization, dan transaction.
    \item Advanced SQL mencakup stored procedure, trigger, JDBC/ODBC, prepared statement, recursive query, window function, cube, dan rollup.
\end{itemize}

\subsection{Database Design using E-R Model}
\begin{itemize}
    \item Fase desain basis data, entity, attribute, relationship, key, cardinality, participation, weak entity, dan E-R diagram.
    \item Extended E-R, UML, isu desain, dan reduksi E-R ke skema relasional.
\end{itemize}

\subsection{Relational Database Design}
\begin{itemize}
    \item Tujuan desain relasional, universal relation, anomaly, functional dependency, closure, key, canonical cover, decomposition, lossless join, dan dependency preservation.
    \item Normal form: 1NF, 2NF, 3NF, BCNF, 4NF, serta trade-off normalisasi, denormalization, dan temporal data modeling.
\end{itemize}

\subsection{Integrity Constraints}
\begin{itemize}
    \item Domain, entity, referential integrity, SQL constraints, named constraints, foreign key action, deferred checking, assertions, triggers, functions, dan procedures.
\end{itemize}

\subsection{Model Data Multidimensi \& Data Warehouse}
\begin{itemize}
    \item OLTP vs OLAP, data warehouse, ETL/ELT, fact, measure, dimension, member, data cube, grain, star schema, snowflake schema, galaxy schema, dan OLAP operations.
\end{itemize}
